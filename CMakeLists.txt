cmake_minimum_required(VERSION 3.20)
project(qbox VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
find_package(Threads REQUIRED)
add_library(qbox_core src/audit.cpp src/config.cpp src/platform.cpp src/protocol.cpp src/payload.cpp src/workspace.cpp src/terminal.cpp)
target_include_directories(qbox_core PUBLIC src)
target_link_libraries(qbox_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(qbox_core PUBLIC ws2_32 bcrypt advapi32)
  target_compile_definitions(qbox_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
endif()
if(MSVC)
  target_compile_options(qbox_core PRIVATE /W4)
else()
  target_compile_options(qbox_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(qbox src/launcher.cpp)
target_link_libraries(qbox PRIVATE qbox_core)
if(MINGW)
  target_link_options(qbox PRIVATE -municode)
endif()
add_executable(qbox-connector src/connector.cpp)
target_link_libraries(qbox-connector PRIVATE qbox_core)
if(MINGW)
  target_link_options(qbox-connector PRIVATE -municode)
endif()
install(TARGETS qbox qbox-connector RUNTIME DESTINATION bin)
option(QBOX_STATIC_GUEST "Statically link the Linux guest runner" ON)
if(UNIX)
  add_executable(runnerd guest/runnerd.c guest/workspace.c guest/rootfs.c guest/audit.c)
  target_compile_options(runnerd PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND QBOX_STATIC_GUEST)
    target_link_options(runnerd PRIVATE -static)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(audit_tests tests/audit_tests.cpp)
  target_link_libraries(audit_tests PRIVATE qbox_core)
  add_test(NAME audit COMMAND audit_tests)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE qbox_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(workspace_tests tests/workspace_tests.cpp)
  target_link_libraries(workspace_tests PRIVATE qbox_core)
  add_test(NAME workspace COMMAND workspace_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME connector_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/connector_policy.py $<TARGET_FILE:qbox-connector>)
  if(UNIX)
    add_executable(workspace_fixture tests/workspace_fixture.c guest/workspace.c)
    add_test(NAME workspace_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/workspace_protocol.py $<TARGET_FILE:workspace_fixture>)
    add_test(NAME runner_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runner_protocol.py $<TARGET_FILE:runnerd>)
    add_test(NAME agent_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/agent_protocol.py $<TARGET_FILE:runnerd>)
    add_test(NAME audit_lifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/audit_lifecycle.py $<TARGET_FILE:qbox>)
    add_test(NAME launcher_lifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/launcher_lifecycle.py $<TARGET_FILE:qbox> $<TARGET_FILE:qbox-connector>)
  endif()
endif()
